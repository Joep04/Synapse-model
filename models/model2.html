#!/usr/bin/env python3 """ Interactive inverted-U model: dopamine, working memory and methylphenidate (MPH). Opens a window with profile tabs at the top (Typical, ADHD, Anxiety or stress) and three sliders: Person: baseline dopamine, extra boost Methylphenidate: plasma concentration (ng/mL) The individual optimum and curve width are fixed (see OPTIMUM and CURVE_WIDTH). Install (once): pip install numpy matplotlib Run: python dopamine_mph_model.py Options: --profile typical|adhd|anxiety starting profile (default: adhd) --mph 5 starting concentration in ng/mL --no-window only save a PNG (no window) --save file.png name of the PNG Scale: relative dopamine from 0 to 4. A typical person sits at 2.0, which is also their optimum. This is an illustrative model, not a diagnostic or medical tool. """ import argparse import sys import textwrap try: import numpy as np import matplotlib import matplotlib.pyplot as plt from matplotlib.widgets import Slider, Button except ImportError: sys.exit("numpy and matplotlib are required. Install them with:\n" " pip install numpy matplotlib") DA_MAX = 4.0 OPTIMUM = 2.0 # dopamine level where working memory peaks CURVE_WIDTH = 0.64 # tolerance of the curve (smaller = narrower peak) # --------------------------------------------------------------------------- # 1. MPH function (unchanged apart from the language) # --------------------------------------------------------------------------- def calculate_dopamine_shift(concentration_mph, baseline_da=1.0, e_max=1.0, ec50=4.5, max_da_increase=2.5): """ Calculates dopamine output from the methylphenidate concentration using an Emax / Hill saturation curve. :param concentration_mph: Plasma concentration in ng/mL :param baseline_da: Default dopamine level without medication (e.g. 1.0) :param e_max: Maximum occupancy fraction (1.0 = 100%) :param ec50: Concentration for 50% DAT occupancy (average ~4.5 ng/mL) :param max_da_increase: Maximum factor by which dopamine can increase :return: The estimated current dopamine level """ # Step 1: calculate DAT occupancy (0.0 to 1.0) dat_occupancy = (e_max * concentration_mph) / (ec50 + concentration_mph) # Step 2: translate occupancy into a dopamine increase current_da = baseline_da + (max_da_increase * dat_occupancy) return current_da # --------------------------------------------------------------------------- # 2. Model # --------------------------------------------------------------------------- # dl = added to the starting level, db = added to the boost, # wm = multiplier of the curve width PROFILES = { "typical": { "name": "Typical", "color": "#898781", "dl": 0.0, "db": 0.0, "wm": 1.0, "text": "No profile effect: only the sliders count. Without " "medication this person sits at the optimum, so extra " "stimulation is more likely to hurt than help."}, "adhd": { "name": "ADHD", "color": "#2a78d6", "dl": -1.0, "db": 0.0, "wm": 1.0, "text": "Lower prefrontal catecholamine signaling: the starting level " "shifts down onto the left slope of the curve. Raise the MPH " "concentration to see medication move the person toward the " "optimum and then past it."}, "anxiety": { "name": "Anxiety or stress", "color": "#c94f8a", "dl": 0.2, "db": 0.6, "wm": 0.875, "text": "Anxiety adds a stress surge of catecholamines and reduces " "tolerance. The person already sits close to or past the " "optimum, so extra stimulation quickly turns negative. " "Norepinephrine matters at least as much as dopamine here, " "so this is a simplification."}, } DEFAULTS = dict(baseline=2.0, boost=0.0, mph=5.0) # Fixed MPH parameters (no sliders); adjust here if you want MPH_PARAMS = dict(ec50=4.5, e_max=1.0, max_inc=2.5) def performance(da, optimum, width): """Working memory performance (0-100%) as a Gaussian around the optimum.""" return 100.0 * np.exp(-((da - optimum) ** 2) / (2.0 * width ** 2)) def zone(da, optimum, width): d = da - optimum if abs(d) <= 0.5 * width: return "Near optimum" return "Under-stimulated" if d < 0 else "Over-stimulated" def compute(baseline, boost, profile, conc): """ Combines person + profile + MPH concentration (conc may be a numpy array). The effective starting level is passed to the MPH function as baseline_da. """ start = float(np.clip(baseline + profile["dl"], 0.0, DA_MAX)) da_mph = calculate_dopamine_shift( conc, baseline_da=start, e_max=MPH_PARAMS["e_max"], ec50=MPH_PARAMS["ec50"], max_da_increase=MPH_PARAMS["max_inc"]) after = np.clip(da_mph + boost + profile["db"], 0.0, DA_MAX) occupancy = (MPH_PARAMS["e_max"] * conc) / (MPH_PARAMS["ec50"] + conc) width = CURVE_WIDTH * profile["wm"] return {"start": start, "after": after, "occupancy": occupancy, "width": width, "p0": performance(start, OPTIMUM, width), "p1": performance(after, OPTIMUM, width)} # --------------------------------------------------------------------------- # 3. Figure with tabs and sliders # --------------------------------------------------------------------------- def build_figure(start_profile="adhd", start_mph=5.0): fig = plt.figure(figsize=(14, 10)) ax1 = fig.add_axes([0.06, 0.35, 0.40, 0.54]) ax2 = fig.add_axes([0.56, 0.35, 0.41, 0.54]) manager = getattr(fig.canvas, "manager", None) if manager is not None: manager.set_window_title("Dopamine, working memory and MPH") t_metrics = fig.text(0.03, 0.285, "", fontsize=12, fontweight="bold", va="top") t_text = fig.text(0.03, 0.245, "", fontsize=10, va="top") def make_slider(rect, label, lo, hi, init, step, fmt="%.2f"): return Slider(fig.add_axes(rect), label, lo, hi, valinit=init, valstep=step, valfmt=fmt) sliders = { "baseline": make_slider([0.20, 0.13, 0.26, 0.03], "Baseline dopamine", 0.0, 4.0, DEFAULTS["baseline"], 0.01), "boost": make_slider([0.20, 0.07, 0.26, 0.03], "Extra boost", -1.2, 1.2, DEFAULTS["boost"], 0.05), "mph": make_slider([0.68, 0.13, 0.26, 0.03], "MPH (ng/mL)", 0.0, 40.0, start_mph, 0.5, "%.1f"), } # Profiles as tab buttons at the top (the active tab is colored) state = {"profile": start_profile} tabs = {} width, gap, x0 = 0.15, 0.01, 0.06 for n, (key, pr) in enumerate(PROFILES.items()): rect = [x0 + n * (width + gap), 0.948, width, 0.04] tabs[key] = Button(fig.add_axes(rect), pr["name"]) reset_button = Button(fig.add_axes([0.68, 0.06, 0.12, 0.04]), "Reset") def update(_=None): baseline = sliders["baseline"].val boost = sliders["boost"].val conc = sliders["mph"].val key = state["profile"] p = PROFILES[key] s = compute(baseline, boost, p, conc) # Left: the inverted U ax1.clear() x = np.linspace(0, DA_MAX, 300) ax1.plot(x, performance(x, OPTIMUM, s["width"]), color=p["color"], lw=3) ax1.plot([s["start"], s["after"]], [s["p0"], s["p1"]], color="#898781", ls="--", lw=1.4) ax1.scatter(s["start"], s["p0"], s=140, color="#eb6834", marker="o", zorder=3, label="Starting level") ax1.scatter(s["after"], s["p1"], s=160, color="#1baf7a", marker="^", zorder=3, label="After MPH and boost") ax1.set_xlim(0, DA_MAX) ax1.set_ylim(0, 105) ax1.set_xlabel("Dopamine (relative units)", fontsize=11) ax1.set_ylabel("Working memory performance (%)", fontsize=11) ax1.set_title("Inverted U: dopamine and performance", fontsize=13) ax1.legend(loc="upper left", frameon=False, fontsize=10) # Right: dose-response for the selected profile only ax2.clear() concs = np.linspace(0, 40, 161) ax2.plot(concs, compute(baseline, boost, p, concs)["p1"], color=p["color"], lw=3) ax2.axvline(conc, color="#898781", ls=":", lw=1.2) ax2.scatter(conc, s["p1"], s=160, color="#1baf7a", marker="^", zorder=3) ax2.set_xlim(0, 40) ax2.set_ylim(0, 105) ax2.set_xlabel("MPH concentration (ng/mL)", fontsize=11) ax2.set_ylabel("Working memory performance (%)", fontsize=11) ax2.set_title(f"Dose-response: {p['name']}", fontsize=13) change = float(s["p1"] - s["p0"]) t_metrics.set_text( f"{p['name']}: performance {float(s['p0']):.0f}% -> " f"{float(s['p1']):.0f}% ({change:+.0f} points) | " f"DAT occupancy {float(s['occupancy']) * 100:.0f}% | " f"{zone(float(s['after']), OPTIMUM, s['width'])}") t_text.set_text(textwrap.fill(p["text"], width=170)) # recolor the tabs for k, b in tabs.items(): active = k == key color = PROFILES[k]["color"] if active else "#e6e5df" b.color = b.hovercolor = color b.ax.set_facecolor(color) b.label.set_color("white" if active else "#2c2c2a") b.label.set_fontweight("bold" if active else "normal") fig.canvas.draw_idle() def choose(key): state["profile"] = key update() def reset(_): sliders["baseline"].set_val(DEFAULTS["baseline"]) sliders["boost"].set_val(DEFAULTS["boost"]) sliders["mph"].set_val(start_mph) for s in sliders.values(): s.on_changed(update) for k, b in tabs.items(): b.on_clicked(lambda _e, k=k: choose(k)) reset_button.on_clicked(reset) update() # return references so the widgets are not garbage collected return fig, (sliders, tabs, reset_button, choose) # --------------------------------------------------------------------------- # 4. Main program # --------------------------------------------------------------------------- def main(): ap = argparse.ArgumentParser(description="Interactive dopamine/MPH model") ap.add_argument("--profile", choices=PROFILES, default="adhd") ap.add_argument("--mph", type=float, default=5.0) ap.add_argument("--no-window", action="store_true") ap.add_argument("--save", default="dopamine_mph_model.png") # parse_known_args: also works in Spyder/Jupyter/IDLE with odd argv args, _ = ap.parse_known_args() print("Example of the MPH function (baseline 1.0):") for conc in [0, 2, 5, 10, 20, 40]: print(f"MPH concentration: {conc:2d} ng/mL -> Estimated dopamine " f"level: {calculate_dopamine_shift(conc):.2f}") fig, _widgets = build_figure(args.profile, args.mph) non_interactive = matplotlib.get_backend().lower() in ( "agg", "pdf", "svg", "ps", "cairo", "template") if args.no_window or non_interactive: fig.savefig(args.save, dpi=130) reason = "requested" if args.no_window else \ "no graphical backend available" print(f"\nFigure saved as {args.save} ({reason}).") return try: plt.show() except Exception as e: # e.g. missing tkinter/Qt fig.savefig(args.save, dpi=130) print(f"\nCould not open a window ({e}).") print(f"Figure saved as {args.save}. Tip: pip install PyQt5, " "or install tkinter.") if __name__ == "__main__": main()
